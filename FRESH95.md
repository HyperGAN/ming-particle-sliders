    fixture=False)

training = open_training(config, cache, new_output_directory,
    mode="fresh", cohort_id=new_cohort, base_identity=base_identity,
    device=owned_device, component_factory=make_original_model,
    owner_admission=admit, fixture=False, ema_decay=None)
